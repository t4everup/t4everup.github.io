{{/* 分页导航。传入 .Paginator（或 .Paginate 的返回值）。 页数少时全部展开；页数多时保留首尾页 + 当前页附近，中间用省略号： 1 2 3 … 12 / 1 … 5 [6] 7 … 12 / 1 … 10 11 12 */}} {{- $p := . -}} {{- if gt $p.TotalPages 1 -}} {{- $current := $p.PageNumber -}} {{- $total := $p.TotalPages -}} {{- $showAll := le $total 7 -}} {{- $start := sub $current 1 -}} {{- $end := add $current 1 -}} {{- if le $current 2 -}} {{- $start = 1 -}} {{- $end = 3 -}} {{- else if ge $current (sub $total 1) -}} {{- $start = sub $total 2 -}} {{- $end = $total -}} {{- end -}}
{{- if $p.HasPrev }} ← 上一页 {{- else }} ← 上一页 {{- end }} {{- range $p.Pagers }} {{- $n := .PageNumber }} {{- $show := or $showAll (eq $n 1) (eq $n $total) (and (ge $n $start) (le $n $end)) }} {{- if $show }} {{- if eq $n $current }} {{ $n }} {{- else }} {{ $n }} {{- end }} {{- else if or (eq $n (sub $start 1)) (eq $n (add $end 1)) }} … {{- end }} {{- end }} {{- if $p.HasNext }} 下一页 → {{- else }} 下一页 → {{- end }}
{{- end -}}
